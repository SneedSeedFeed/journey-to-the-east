jtte_has_shih_mentor_trigger = {
    OR = {
        any_relation = { 
            type = friend 
            is_alive = yes 
            has_trait = shih 
        }
        any_relation = { 
            type = best_friend 
            is_alive = yes 
            has_trait = shih 
        }
        any_relation = { 
            type = lover 
            is_alive = yes 
            has_trait = shih 
        }
        any_relation = { 
            type = soulmate 
            is_alive = yes 
            has_trait = shih 
        }
    }
}

jtte_can_train_trigger = {
    is_ai = no
    is_adult = yes
    POD_is_supehunter_trigger = yes
    NOT = { has_trait = shih }
    NOT = { has_character_flag = qiao_access }
    has_character_flag = jtte_training_started
}

jtte_can_split_trigger = {
    is_ai = no
    has_trait = hun_and_po
    NOT = { has_character_flag = jtte_soul_split_complete }
}

jtte_has_research_text_trigger = {
    custom_tooltip = {
        text = jtte_qiao_research_text_tt
        any_character_artifact = {
            OR = {
                has_variable = baizetu
                has_artifact_modifier = manual_of_yi_modifier
            }
        }
    }
}
