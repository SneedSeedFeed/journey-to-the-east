jtte_purpose_attempt_effect = {
    
    hidden_effect = { 
        set_variable = { 
            name = jtte_purpose_approach 
            value = flag:$APPROACH$ 
        } 
    }

    duel = {
        skill = $SKILL$
        value = 10
        30 = {
            desc = jtte.purpose.$APPROACH$.comfortable
            compare_modifier = {
                value = scope:duel_value
                multiplier = 2
                min = -29
                max = 29
            }
            hidden_effect = { 
                set_variable = { 
                    name = jtte_purpose_result 
                    value = flag:comfortable 
                } 
            }
            custom_tooltip = jtte_purpose_no_stress_tt
        }
        40 = {
            desc = jtte.purpose.$APPROACH$.strained
            hidden_effect = { 
                set_variable = { 
                    name = jtte_purpose_result 
                    value = flag:strained 
                } 
            }
            show_as_tooltip = { add_stress = 40 }
        }
        30 = {
            desc = jtte.purpose.$APPROACH$.barely
            compare_modifier = {
                value = scope:duel_value
                multiplier = -2
                min = -29
                max = 29
            }
            hidden_effect = { set_variable = { name = jtte_purpose_result value = flag:barely } }
            show_as_tooltip = { add_stress = 80 }
        }
    }
    trigger_event = { id = jtte.201 days = 1 }
}
